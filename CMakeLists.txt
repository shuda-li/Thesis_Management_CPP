cmake_minimum_required(VERSION 3.16)
project(ThesisFlow VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(sqlite3 STATIC third_party/sqlite3.c)
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
add_executable(thesis_manager src/main.cpp src/app.cpp)
target_include_directories(thesis_manager PRIVATE include third_party)
target_link_libraries(thesis_manager PRIVATE sqlite3 Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
 target_link_libraries(thesis_manager PRIVATE ws2_32 bcrypt)
endif()
if(MSVC)
 target_compile_options(thesis_manager PRIVATE /utf-8 /W4)
else()
 target_compile_options(thesis_manager PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(security_test tests/security_test.cpp)
target_include_directories(security_test PRIVATE include third_party)
if(WIN32)
 target_link_libraries(security_test PRIVATE bcrypt)
endif()
enable_testing()
add_test(NAME security_vectors COMMAND security_test)
